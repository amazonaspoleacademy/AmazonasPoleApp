from pathlib import Path import base64, re, subprocess html_src = Path("/mnt/data/index_amazonas_logo_real.html") logo_path = Path("/mnt/data/ChatGPT Image 20 ago 2026, 02_07_13.png") out = Path("/mnt/data/index_amazonas_logo_circular.html") text = html_src.read_text(encoding="utf-8") logo_b64 = base64.b64encode(logo_path.read_bytes()).decode("ascii") data_uri = f"data:image/png;base64,{logo_b64}" # Replace all embedded PNG data URIs used for the previous logo with the new circular logo. text = re.sub(r'data:image/png;base64,[A-Za-z0-9+/=]+', data_uri, text) out.write_text(text, encoding="utf-8") # Validate JS syntax. m = re.search(r"", text, re.S) if not m: raise RuntimeError("No se encontró el bloque script.") check = Path("/mnt/data/check_logo_circular.js") check.write_text(m.group(1), encoding="utf-8") res = subprocess.run(["node", "--check", str(check)], capture_output=True, text=True) print("Archivo creado:", out) print("Sintaxis JavaScript:", "OK" if res.returncode == 0 else res.stderr) print("Tamaño:", out.stat().st_size, "bytes")
